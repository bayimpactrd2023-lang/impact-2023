-- News table indexes
-- Speeds up ordering by date (most common query)
CREATE INDEX IF NOT EXISTS idx_news_date
ON news(date DESC);

-- =====================================================

-- Highlights table indexes
-- Speeds up ordering by published date
CREATE INDEX IF NOT EXISTS idx_highlights_published_date
ON highlights(published_date DESC NULLS LAST);

-- Speeds up filtering by featured status
CREATE INDEX IF NOT EXISTS idx_highlights_featured
ON highlights(featured) WHERE featured = true;

-- Composite index for featured + published date
CREATE INDEX IF NOT EXISTS idx_highlights_featured_date
ON highlights(featured, published_date DESC NULLS LAST);

-- =====================================================

-- Publications table indexes
-- Speeds up ordering by published date
CREATE INDEX IF NOT EXISTS idx_publications_published_date
ON publications(published_date DESC NULLS LAST);

-- Speeds up filtering by featured status
CREATE INDEX IF NOT EXISTS idx_publications_featured
ON publications(featured) WHERE featured = true;

-- =====================================================

-- Blog posts table indexes
-- Speeds up ordering by date
CREATE INDEX IF NOT EXISTS idx_blog_posts_date
ON blog_posts(date DESC);

-- Speeds up filtering by author
CREATE INDEX IF NOT EXISTS idx_blog_posts_author
ON blog_posts(author);

-- Composite index for date + author
CREATE INDEX IF NOT EXISTS idx_blog_posts_date_author
ON blog_posts(date DESC, author);

-- =====================================================

-- Projects table indexes
-- Speeds up filtering by category
CREATE INDEX IF NOT EXISTS idx_projects_category
ON projects(category);

-- Speeds up ordering by date
CREATE INDEX IF NOT EXISTS idx_projects_date
ON projects(date DESC NULLS LAST);

-- Composite index for category + date
CREATE INDEX IF NOT EXISTS idx_projects_category_date
ON projects(category, date DESC NULLS LAST);

-- =====================================================

-- Financial statements table indexes
-- Speeds up ordering by year (most recent first)
CREATE INDEX IF NOT EXISTS idx_financial_year
ON financial_statements(year DESC);

-- =====================================================

-- Team members table indexes
-- Speeds up ordering by name
CREATE INDEX IF NOT EXISTS idx_team_name
ON team_members(name);

-- Speeds up filtering by role
CREATE INDEX IF NOT EXISTS idx_team_role
ON team_members(role);

-- =====================================================

-- Partners table indexes
-- Speeds up ordering by name
CREATE INDEX IF NOT EXISTS idx_partners_name
ON partners(name);

-- =====================================================

-- Internship testimonials table indexes
-- Speeds up ordering by published date
CREATE INDEX IF NOT EXISTS idx_testimonials_published_date
ON internship_testimonials(published_date DESC NULLS LAST);

-- Speeds up filtering by year
CREATE INDEX IF NOT EXISTS idx_testimonials_year
ON internship_testimonials(year DESC);

-- Composite index for year + published date
CREATE INDEX IF NOT EXISTS idx_testimonials_year_date
ON internship_testimonials(year DESC, published_date DESC NULLS LAST);

-- =====================================================

-- Hero sections table indexes
-- Speeds up ordering by created_at
CREATE INDEX IF NOT EXISTS idx_hero_created
ON hero_sections(created_at DESC);

-- =====================================================

-- VERIFICATION QUERIES
-- Run these to check if indexes were created successfully
-- =====================================================

-- List all indexes
SELECT
  schemaname,
  tablename,
  indexname,
  indexdef
FROM pg_indexes
WHERE schemaname = 'public'
ORDER BY tablename, indexname;

-- Check index sizes
SELECT
  schemaname || '.' || tablename AS table,
  indexname,
  pg_size_pretty(pg_relation_size(indexrelid)) AS index_size
FROM pg_stat_user_indexes
WHERE schemaname = 'public'
ORDER BY pg_relation_size(indexrelid) DESC;
