-- =====================================================
-- PRODUCTION DATABASE INDEXES
-- =====================================================
-- Purpose: Optimize database query performance
-- Impact: 2-10x faster queries on large datasets
-- Run in: Supabase SQL Editor
-- Date: March 29, 2026
-- =====================================================

-- News table indexes
-- Speeds up ordering by date (most common query)
CREATE INDEX IF NOT EXISTS idx_news_date 
ON news(date DESC);

-- Speeds up filtering by category
CREATE INDEX IF NOT EXISTS idx_news_category 
ON news(category);

-- Composite index for date + category queries
CREATE INDEX IF NOT EXISTS idx_news_date_category 
ON news(date DESC, category);

-- =====================================================

-- Highlights table indexes
-- Speeds up ordering by published date
CREATE INDEX IF NOT EXISTS idx_highlights_published_date 
ON highlights(published_date DESC NULLS LAST);

-- Speeds up filtering by featured status
CREATE INDEX IF NOT EXISTS idx_highlights_is_featured 
ON highlights(is_featured) WHERE is_featured = true;

-- Composite index for featured + published date
CREATE INDEX IF NOT EXISTS idx_highlights_featured_date 
ON highlights(is_featured, published_date DESC NULLS LAST);

-- =====================================================

-- Publications table indexes
-- Speeds up ordering by published date
CREATE INDEX IF NOT EXISTS idx_publications_published_date 
ON publications(published_date DESC NULLS LAST);

-- Speeds up filtering by type
CREATE INDEX IF NOT EXISTS idx_publications_type 
ON publications(type);

-- Speeds up filtering by featured status
CREATE INDEX IF NOT EXISTS idx_publications_is_featured 
ON publications(is_featured) WHERE is_featured = true;

-- Composite index for type + published date
CREATE INDEX IF NOT EXISTS idx_publications_type_date 
ON publications(type, published_date DESC NULLS LAST);

-- =====================================================

-- Blog posts table indexes
-- Speeds up ordering by date
CREATE INDEX IF NOT EXISTS idx_blog_posts_date 
ON blog_posts(date DESC);

-- Speeds up filtering by category
CREATE INDEX IF NOT EXISTS idx_blog_posts_category 
ON blog_posts(category);

-- Composite index for date + category
CREATE INDEX IF NOT EXISTS idx_blog_posts_date_category 
ON blog_posts(date DESC, category);

-- =====================================================

-- Projects table indexes
-- Speeds up filtering by category
CREATE INDEX IF NOT EXISTS idx_projects_category 
ON projects(category);

-- Speeds up filtering by funding type
CREATE INDEX IF NOT EXISTS idx_projects_funding_type 
ON projects(funding_type);

-- Composite index for category + funding type
CREATE INDEX IF NOT EXISTS idx_projects_category_funding 
ON projects(category, funding_type);

-- =====================================================

-- Financial statements table indexes
-- Speeds up ordering by year (most recent first)
CREATE INDEX IF NOT EXISTS idx_financial_year 
ON financial_statements(year DESC);

-- Speeds up filtering by type
CREATE INDEX IF NOT EXISTS idx_financial_type 
ON financial_statements(type);

-- =====================================================

-- Team members table indexes
-- Speeds up ordering by position/role
CREATE INDEX IF NOT EXISTS idx_team_position 
ON team_members(position);

-- Speeds up filtering by department
CREATE INDEX IF NOT EXISTS idx_team_department 
ON team_members(department) WHERE department IS NOT NULL;

-- =====================================================

-- Partners table indexes
-- Speeds up filtering by type
CREATE INDEX IF NOT EXISTS idx_partners_type 
ON partners(type);

-- =====================================================

-- Internship testimonials table indexes
-- Speeds up ordering by published date
CREATE INDEX IF NOT EXISTS idx_testimonials_published_date 
ON internship_testimonials(published_date DESC NULLS LAST);

-- =====================================================

-- Hero slides table indexes
-- Speeds up ordering by order field
CREATE INDEX IF NOT EXISTS idx_hero_order 
ON hero_slides("order" ASC);

-- =====================================================

-- VERIFICATION QUERIES
-- Run these to check if indexes were created successfully
-- =====================================================

-- List all indexes
SELECT 
  schemaname,
  tablename,
  indexname,
  indexdef
FROM pg_indexes
WHERE schemaname = 'public'
ORDER BY tablename, indexname;

-- Check index sizes
SELECT
  schemaname || '.' || tablename AS table,
  indexname,
  pg_size_pretty(pg_relation_size(indexrelid)) AS index_size
FROM pg_stat_user_indexes
WHERE schemaname = 'public'
ORDER BY pg_relation_size(indexrelid) DESC;
